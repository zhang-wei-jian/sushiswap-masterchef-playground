function deposit(uint256 _amount) public {
    PoolInfo storage pool = poolInfo[0];
    UserInfo storage user = userInfo[msg.sender];
    updatePool(); // 1. 同步全局水位
    if (user.amount > 0) {
        uint256 pending = (user.amount * pool.acc) - user.rewardDebt;
        safeTransfer(msg.sender, pending); // 2. 发放旧账
    }
    user.amount += _amount; // 3. 更新个人股份
    user.rewardDebt = user.amount * pool.acc; // 4. 重新立快照
}

function withdraw(uint256 _amount) public {
    PoolInfo storage pool = poolInfo[0];
    UserInfo storage user = userInfo[msg.sender];
    updatePool(); // 1. 同步全局水位
    uint256 pending = (user.amount * pool.acc) - user.rewardDebt;
    safeTransfer(msg.sender, pending); // 2. 发放旧账
    user.amount -= _amount; // 3. 减少个人股份
    user.rewardDebt = user.amount * pool.acc; // 4. 重新立快照
}

function updatePool() internal {
    uint256 lpSupply = lpToken.balanceOf(address(this));
    uint256 reward = (block.number - lastBlock) * sushiPerBlock;
    pool.acc += (reward * 1e12) / lpSupply; // 累加每股分红
    pool.lastBlock = block.number;
}
